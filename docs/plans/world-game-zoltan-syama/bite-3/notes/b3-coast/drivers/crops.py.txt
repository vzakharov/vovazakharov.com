# Throwaway: side-by-side 2x crops of one region across COAST_FINER 2/1/0, labelled, and the pixel
# difference of each against 2. `python3 -I crops.py <frames dir> <out dir>`
import sys
from PIL import Image, ImageChops, ImageDraw, ImageStat

frames, out = sys.argv[1], sys.argv[2]
REGIONS = {
    '13km': [(880, 0, 1140, 260), (560, 560, 820, 820)],
    '527m': [(300, 380, 820, 640), (980, 80, 1320, 260)],
}
for shot, boxes in REGIONS.items():
    imgs = {cf: Image.open(f'{frames}/cf{cf}-{shot}.png').convert('RGB') for cf in (2, 1, 0)}
    for cf in (1, 0):
        diff = ImageChops.difference(imgs[2], imgs[cf]).convert('L')
        changed = sum(1 for v in diff.getdata() if v > 16)
        print(f'{shot} cf{cf} vs cf2: mean |diff| {ImageStat.Stat(diff).mean[0]:.2f}, px over 16: {changed}')
    rows = []
    for box in boxes:
        w, h = box[2] - box[0], box[3] - box[1]
        row = Image.new('RGB', (3 * 2 * w + 2 * 8, 2 * h), 'white')
        for i, cf in enumerate((2, 1, 0)):
            tile = imgs[cf].crop(box).resize((2 * w, 2 * h), Image.NEAREST)
            ImageDraw.Draw(tile).text((8, 8), f'COAST_FINER {cf}', fill='white')
            row.paste(tile, (i * (2 * w + 8), 0))
        rows.append(row)
    sheet = Image.new('RGB', (max(r.width for r in rows), sum(r.height for r in rows) + 8 * (len(rows) - 1)), 'white')
    y = 0
    for r in rows:
        sheet.paste(r, (0, y))
        y += r.height + 8
    sheet.save(f'{out}/crops-{shot}.png')
