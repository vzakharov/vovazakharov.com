"""count3.py <plain.png> <tint.png> [<overlay.png>]: level-edge px vs coast px of one settled frame.

Level edge: tint colour changes (>60, as edges.py) between two non-sky pixels of the
same sea/land class in the plain frame. Coast: a plain sea pixel next to a plain land
pixel. Overlay: level edges red, coast yellow, over the plain frame.
"""
import sys

import numpy as np
from PIL import Image

plain, tint = sys.argv[1:3]
p = np.array(Image.open(plain).convert('RGB')).astype(int)
t = np.array(Image.open(tint).convert('RGB')).astype(int)
sky = (np.abs(p - t).sum(2) < 12) & (p[..., 2] > 200) & (p[..., 0] > 140) & (p[..., 0] < 200)
r, g, b = p[..., 0], p[..., 1], p[..., 2]
sea = ~sky & (b > r + 40) & (b > g + 10)


def pairs(a, b_):
    return [(a[1:], a[:-1], b_[1:], b_[:-1]), (a[:, 1:], a[:, :-1], b_[:, 1:], b_[:, :-1])]


lvl = np.zeros(sky.shape, bool)
coast = np.zeros(sky.shape, bool)
for axis in (0, 1):
    sl1 = (slice(1, None), slice(None)) if axis == 0 else (slice(None), slice(1, None))
    sl0 = (slice(None, -1), slice(None)) if axis == 0 else (slice(None), slice(None, -1))
    ground = ~sky[sl1] & ~sky[sl0]
    same = sea[sl1] == sea[sl0]
    tchg = np.abs(t[sl1] - t[sl0]).sum(2) > 60
    lvl[sl0] |= ground & same & tchg
    coast[sl0] |= ground & ~same
ground_px = int((~sky).sum())
print(f'{plain.split("/")[-1]}\tground {ground_px}\tlevel-edge {int(lvl.sum())}\tcoast {int(coast.sum())}\tsea {int(sea.sum())}')
if len(sys.argv) > 3:
    o = p.copy()
    o[coast] = [255, 220, 0]
    o[lvl] = [255, 0, 0]
    Image.fromarray(o.astype('uint8')).save(sys.argv[3])
