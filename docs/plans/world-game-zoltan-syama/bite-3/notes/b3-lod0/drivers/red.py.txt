"""Counts stand-in pixels in a tinted build (red over green) per frame, whole frame."""
import sys
from pathlib import Path

from PIL import Image

for f in sorted(Path(sys.argv[1]).glob("*.png")):
    im = Image.open(f).convert("RGB")
    w, h = im.size
    px = im.load()
    below = float(sys.argv[2]) if len(sys.argv) > 2 else 0.0
    n = low = 0
    for y in range(h):
        for x in range(w):
            r, g, b = px[x, y]
            if r > 40 and r > 1.35 * g and b < 0.5 * r:
                n += 1
                low += y >= below * h
    print(f"{f.stem}\t{n}\t{low}")
