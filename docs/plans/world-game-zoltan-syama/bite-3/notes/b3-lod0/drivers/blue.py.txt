"""Counts hole-coloured pixels (the globe's flat sea, ~(5,34,66)) per frame, whole frame and below a row."""
import sys
from pathlib import Path

from PIL import Image

HOLE = (7, 41, 78)
TOL = 4
below = float(sys.argv[2]) if len(sys.argv) > 2 else 0.0
for f in sorted(Path(sys.argv[1]).glob("*.png")):
    im = Image.open(f).convert("RGB")
    w, h = im.size
    px = im.load()
    total = low = 0
    for y in range(0, h):
        for x in range(0, w):
            r, g, b = px[x, y]
            if abs(r - HOLE[0]) <= TOL and abs(g - HOLE[1]) <= TOL and abs(b - HOLE[2]) <= TOL:
                total += 1
                if y >= below * h:
                    low += 1
    print(f"{f.stem}\t{total}\t{low}")
