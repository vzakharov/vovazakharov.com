// For each reported cluster: where its outside imports land, whether the rest
// of the slice imports it, and who enters the slice from outside.
import { readFileSync } from 'node:fs';
import path from 'node:path';
import ts from 'typescript';
import { appFiles, resolveImport, sourceFile } from '../../scripts/lib/import-graph.ts';

const loc = (f: string) => {
  const p = f.split('/');
  if (p[0] === 'sites') return 'router';
  if (['entities', 'features', 'widgets', 'pages'].includes(p[2])) return p.slice(1, 4).join('/');
  return p.slice(1, 3).join('/') + (p[2] === 'shared' ? '/' + p[3] : '');
};
const test = (f: string) => /\.test(-kit)?\.tsx?$|index\.test-only\.ts$/.test(f);
const out = new Map<string, string[]>();
for (const f of appFiles()) {
  if (test(f)) continue;
  const t: string[] = [];
  for (const { fileName } of ts.preProcessFile(readFileSync(f, 'utf8'), true, true).importedFiles) {
    const r = resolveImport(f, fileName);
    const s = r && sourceFile(r);
    if (s && s !== f && !test(s)) t.push(s);
  }
  out.set(f, t);
}
const rep = readFileSync('tmp/cluster-moves/report.txt', 'utf8').split('\n');
let slice = '';
const clusters: { slice: string; root: string; files: string[] }[] = [];
for (const l of rep) {
  let m;
  if ((m = /^src\/(\S+) /.exec(l))) slice = m[1];
  else if ((m = /^  (\S+): /.exec(l))) clusters.push({ slice, root: m[1], files: [] });
  else if ((m = /^    (\S+)/.exec(l))) clusters.at(-1)!.files.push(`src/${slice}/${m[1]}`);
}
for (const c of clusters) {
  const T = new Set(c.files);
  const ext = new Set<string>();
  for (const f of c.files) for (const g of out.get(f) ?? []) if (loc(g) !== c.slice) ext.add(loc(g));
  const restIn = [...out]
    .filter(([f, gs]) => loc(f) === c.slice && !T.has(f) && gs.some((g) => T.has(g)))
    .map(([f]) => path.basename(f));
  const entering = new Set(
    [...out].filter(([f, gs]) => loc(f) !== c.slice && gs.some((g) => loc(g) === c.slice)).map(([f]) => loc(f)),
  );
  console.log(
    `${c.slice} :: ${c.root} (${c.files.length})\n  imports: ${[...ext].sort().join(', ') || '—'}\n  rest importing it: ${restIn.join(', ') || '—'}\n  slice entered by: ${[...entering].sort().join(', ')}`,
  );
}
