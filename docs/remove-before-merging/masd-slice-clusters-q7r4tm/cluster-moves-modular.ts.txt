// check:masd's test, e − a² over the app's own edges, applied to each reported
// cluster as if it were a slice of its own.
import { readFileSync } from 'node:fs';
import ts from 'typescript';
import { appFiles, resolveImport, sourceFile } from '../../scripts/lib/import-graph.ts';

const test = (f: string) => /\.test(-kit)?\.tsx?$|index\.test-only\.ts$/.test(f);
const appOf = (f: string) => {
  const p = f.split('/');
  return p[0] === 'src' && p.length >= 4 && p[2] !== 'scripts' && !test(f) ? p[1] : undefined;
};
const edges = new Map<string, [string, string][]>();
for (const f of appFiles()) {
  const app = appOf(f);
  if (!app) continue;
  const linked = new Set<string>();
  for (const { fileName } of ts.preProcessFile(readFileSync(f, 'utf8'), true, true).importedFiles) {
    const r = resolveImport(f, fileName);
    const s = r && sourceFile(r);
    if (s && s !== f && appOf(s) === app) linked.add(s);
  }
  const own = edges.get(app) ?? [];
  own.push(...[...linked].map((s): [string, string] => [f, s]));
  edges.set(app, own);
}
let slice = '';
let root = '';
let files: string[] = [];
const flush = () => {
  if (!root) return;
  const app = slice.split('/')[0];
  const es = edges.get(app) ?? [];
  const T = new Set(files);
  let inside = 0;
  let ends = 0;
  for (const [a, b] of es) {
    const n = Number(T.has(a)) + Number(T.has(b));
    if (n === 2) inside++;
    ends += n;
  }
  const m = es.length;
  const e = inside / m;
  const a = ends / (2 * m);
  console.log(`${(e - a * a > 0 ? 'pass' : 'FAIL').padEnd(4)} e−a²=${(e - a * a).toFixed(4)} m=${m} ${slice} :: ${root} (${files.length})`);
};
for (const l of readFileSync('tmp/cluster-moves/report.txt', 'utf8').split('\n')) {
  let mm;
  if ((mm = /^src\/(\S+) /.exec(l))) (flush(), (root = ''), (slice = mm[1]));
  else if ((mm = /^  (\S+): /.exec(l))) (flush(), (root = mm[1]), (files = []));
  else if ((mm = /^    (\S+)/.exec(l))) files.push(`src/${slice}/${mm[1]}`);
}
flush();
