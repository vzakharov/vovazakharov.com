// Scratch (not committed): all ten fliers over six mushrooms, a cloud tapped
// 600 frames on, then ~1020 frames; reports what the watch saw.
import fs from 'node:fs';

import { z } from 'zod';

import { Watch } from './flier-watch.ts';
import {
  Clouds,
  type Controls,
  type Expect,
  inTurn,
  type Page,
} from './mushroom-probe.ts';

const State = z.object({ mushrooms: z.array(z.unknown()) });

export async function playZzShower(
  page: Page,
  controls: z.infer<typeof Controls>,
  _expect: Expect,
  note: (line: string) => void,
): Promise<void> {
  const dry = process.env['ZZ_DRY'] === '1';
  const room =
    6 - (await page.evaluate('__probe.state()', State)).mushrooms.length;
  await inTurn([...Array.from({ length: room }).keys()], async (index) => {
    await page.tap(controls.plus);
    await page.step(30);
    const cap =
      controls.picker[index % 2 === 0 ? 0 : index % controls.picker.length];
    if (cap) await page.tap(cap);
    await page.step(60);
  });
  const kinds = [
    ...Array.from({ length: 4 }, () => 'butterfly' as const),
    ...Array.from({ length: 3 }, () => 'fly' as const),
    ...Array.from({ length: 3 }, () => 'bee' as const),
  ];
  await inTurn(kinds, async (kind) => {
    await page.tap(controls.releases[kind]);
    await page.step(12);
  });
  await page.step(600);
  const at = await page.evaluate('__game.now?.() ?? null', z.unknown());
  if (!dry) {
    const cloud = (await page.evaluate('__probe.clouds()', Clouds)).find(
      (point) => point !== null,
    );
    if (cloud) await page.tap(cloud);
    else note('ZZ no cloud');
  }
  const trace: unknown[] = [];
  await inTurn([...Array.from({ length: 150 }).keys()], async () => {
    await page.step(1);
    trace.push(
      await page.evaluate(
        `(() => { const s = __probe.scene; const t = s.meadow.now ?? null; return { clock: __probe.state().clock, b: s.meadow.insects.filter((i) => i.kind === 'butterfly').map((i) => { const sh = s.insects.shown.get(i.id); return { id: i.id, rot: sh?.container.rotation, turn: sh?.steering?.turn, to: i.leg.to.kind, dep: i.leg.departs, arr: i.leg.arrives, piv: i.leg.pivots ?? null, lifted: sh?.steering?.setOff?.turns?.lifted ?? null }; }) }; })()`,
        z.unknown(),
      ),
    );
  });
  fs.writeFileSync(
    `${process.env['ZZ_OUT'] ?? '/tmp/zz'}-${String(Math.random()).slice(2, 6)}.json`,
    JSON.stringify(trace),
  );
  await page.step(870);
  const watch = await page.evaluate('window.__watch', Watch);
  const steps = await page.evaluate(
    'window.__watch.turnSteps',
    z.record(z.string(), z.object({ most: z.number() })),
  );
  note(
    `ZZ ${dry ? 'dry' : 'wet'} tap at ${JSON.stringify(at)}; worstTurn ${JSON.stringify(watch.worstTurn)}; per-kind most step x60 ${Object.entries(
      steps,
    )
      .map(([kind, { most }]) => `${kind} ${(most * 60).toFixed(2)}`)
      .join(', ')}; heading ${watch.worstHeading.off.toFixed(2)}, rest ${watch.worstRest.turn.toFixed(2)}, spin ${watch.worstSpin.spin.toFixed(2)}`,
  );
}
