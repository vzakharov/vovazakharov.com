#!/bin/bash
# run2.sh <name>... ; two-stage fold of each tmp/tg/dilo2/<name>.steps.json in parallel -> runs/dilo2/t_<name>
cd /home/user/vovazakharov.com
for n in "$@"; do
  ( d=tmp/tg/runs/dilo2/t_$n; mkdir -p $d
    out=$(npx tsx tmp/tg/dilo2/two.ts tmp/tg/dilo2/$n.steps.json $d/m.json 2>&1 | grep -E "crossings|Error" | tr '\n' ' ')
    ROT=0 npx tsx tmp/tg/show.ts $d/m.json $d >/dev/null 2>&1
    echo "t_$n: $out" ) &
done
wait
