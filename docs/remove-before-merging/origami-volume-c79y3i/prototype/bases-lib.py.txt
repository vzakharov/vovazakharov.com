def swan(hn=0.5, hh=0.6, hm=0.7, n=0.9, hd=1.6, half=True):
    s=[L((n/2-0.15,n/2+0.15),(n/2+0.15,n/2-0.15),(0.05,0.05),hold=hn)]
    a=hd/2
    s.append(L((a-0.1,a+0.1),(a+0.1,a-0.1),(n-0.02,n-0.02),select={"at":[n-0.02,n-0.02],"top":2},direction="valley",hold=hh))
    if half: s.append(L((0,0),(1,1),(0.6,0.4),direction="mountain",hold=hm))
    return s
