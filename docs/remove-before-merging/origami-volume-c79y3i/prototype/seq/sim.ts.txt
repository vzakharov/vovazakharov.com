/**
 * The relax of `relax.ts` kept alive between phases, so a model can be folded
 * along its sequence: each phase moves every hinge's goal from where the hinge
 * is now to a new target over `ramp` iterations, then lets it settle. Positions,
 * tracked angles and contact (with its friction anchors) carry over.
 */
import { contact, crossings, type Stack } from './contact.ts';
import { dihedral } from '../../../src/origami/shared/fold/hinge.ts';
import type { Mesh } from '../../../src/origami/shared/fold/mesh.ts';
import type { Paper } from '../../../src/origami/shared/fold/paper.ts';

const AXIAL = 20;
const CREASE = Number(process.env.CREASE ?? 0.7);
const FACET = Number(process.env.FACET ?? 0.7);
const CONTACT = 4 * AXIAL;
const CONTACT_LOAD = 12;
const REFRESH = 10;

const unwrap = (angle: number, near: number): number =>
  angle + 2 * Math.PI * Math.round((near - angle) / (2 * Math.PI));

export type Phase = {
  readonly iterations: number;
  readonly settled: boolean;
  readonly force: number;
};

export function simulator(mesh: Mesh, paper: Paper, x0: Float64Array) {
  const n = mesh.paper.length;
  const gap = paper.thickness / paper.side;
  const edge = Math.max(...mesh.bars.map(([, , length]) => length));
  const x = Float64Array.from(x0);
  const v = new Float64Array(3 * n);
  const f = new Float64Array(3 * n);
  const a = new Float64Array(3 * n);
  const gradient = new Float64Array(12);
  const bars = Int32Array.from(mesh.bars.flatMap(([p, q]) => [p, q]));
  const rest = Float64Array.from(mesh.bars.map(([, , length]) => length));
  const hinges = mesh.hinges;
  const corners = Int32Array.from(hinges.flatMap((h) => [h.a, h.b, h.c, h.d]));
  const stiffness = Float64Array.from(hinges.map((h) => (h.crease ? CREASE : FACET) * h.rest));
  const tracked = Float64Array.from(hinges.map((h) => dihedral(x, h)));
  const from = new Float64Array(hinges.length);
  const target = new Float64Array(hinges.length);
  let ramp = 1;
  const wings = { a: 0, b: 0, c: 0, d: 0 };
  const touch = contact(mesh, gap, CONTACT / edge);

  const rubbing = process.env.FRICTION || process.env.BOND ? 1 + Number(process.env.SHEAR ?? 1) : 1;
  const mass = new Float64Array(n).fill((CONTACT_LOAD * CONTACT * rubbing) / edge);
  for (const [p, q, length] of mesh.bars) {
    for (const i of [p, q]) mass[i] = (mass[i] ?? 0) + (2 * AXIAL) / length;
  }
  for (const [h, hinge] of hinges.entries()) {
    dihedral(x, hinge, gradient);
    for (const [k, i] of [hinge.a, hinge.b, hinge.c, hinge.d].entries()) {
      const g2 = (gradient[3 * k] ?? 0) ** 2 + (gradient[3 * k + 1] ?? 0) ** 2 + (gradient[3 * k + 2] ?? 0) ** 2;
      mass[i] = (mass[i] ?? 0) + 4 * (stiffness[h] ?? 0) * g2;
    }
  }
  const maxDt = 1.8;
  const maxStep = gap / 2;

  const forces = (): number => {
    f.fill(0);
    for (let e = 0; e < rest.length; e++) {
      const p = 3 * (bars[2 * e] ?? 0);
      const q = 3 * (bars[2 * e + 1] ?? 0);
      const dx = (x[q] ?? 0) - (x[p] ?? 0);
      const dy = (x[q + 1] ?? 0) - (x[p + 1] ?? 0);
      const dz = (x[q + 2] ?? 0) - (x[p + 2] ?? 0);
      const len = Math.sqrt(dx * dx + dy * dy + dz * dz);
      const length = rest[e] ?? len;
      const pull = (AXIAL * (len - length)) / length / len;
      f[p] += pull * dx;
      f[p + 1] += pull * dy;
      f[p + 2] += pull * dz;
      f[q] -= pull * dx;
      f[q + 1] -= pull * dy;
      f[q + 2] -= pull * dz;
    }
    for (let h = 0; h < target.length; h++) {
      wings.a = corners[4 * h] ?? 0;
      wings.b = corners[4 * h + 1] ?? 0;
      wings.c = corners[4 * h + 2] ?? 0;
      wings.d = corners[4 * h + 3] ?? 0;
      const angle = unwrap(dihedral(x, wings, gradient), tracked[h] ?? 0);
      tracked[h] = angle;
      const goal = (from[h] ?? 0) + ((target[h] ?? 0) - (from[h] ?? 0)) * ramp;
      const torque = -(stiffness[h] ?? 0) * (angle - goal);
      for (let k = 0; k < 4; k++) {
        const i = 3 * (corners[4 * h + k] ?? 0);
        f[i] += torque * (gradient[3 * k] ?? 0);
        f[i + 1] += torque * (gradient[3 * k + 1] ?? 0);
        f[i + 2] += torque * (gradient[3 * k + 2] ?? 0);
      }
    }
    touch.push(x, f);
    let worst = 0;
    for (let i = 0; i < 3 * n; i += 3) {
      worst = Math.max(worst, (f[i] ?? 0) ** 2 + (f[i + 1] ?? 0) ** 2 + (f[i + 2] ?? 0) ** 2);
    }
    return Math.sqrt(worst);
  };

  /** Ramps every hinge's goal from its angle now to `goals`, then settles. */
  const phase = (
    goals: Float64Array,
    rampIterations: number,
    maxIterations: number,
    tolerance: number,
    /** Contact pairs left out until iteration `until`. */
    suspended?: { readonly skip: (v: number, t: number) => boolean; readonly until: number },
    /** Friction for this phase; `FRICTION` when left out. */
    friction = Number(process.env.FRICTION ?? 0),
  ): Phase => {
    touch.rubAt(friction);
    from.set(tracked);
    target.set(goals);
    v.fill(0);
    let dt = maxDt / 10;
    let alpha = 0.1;
    let calm = 0;
    let force = Infinity;
    for (let iteration = 0; iteration < maxIterations; iteration++) {
      ramp = Math.min(1, iteration / rampIterations);
      const held = suspended !== undefined && iteration < suspended.until;
      if (iteration % REFRESH === 0) {
        touch.suspend(held ? suspended.skip : undefined);
        touch.refresh(x, 2 * REFRESH * maxStep);
      }
      force = forces();
      if (force < tolerance && ramp === 1 && !held) return { iterations: iteration, settled: true, force };
      for (let i = 0; i < 3 * n; i++) a[i] = (f[i] ?? 0) / (mass[Math.floor(i / 3)] ?? 1);
      let power = 0;
      let vv = 0;
      let aa = 0;
      for (let i = 0; i < 3 * n; i++) {
        power += (a[i] ?? 0) * (v[i] ?? 0);
        vv += (v[i] ?? 0) ** 2;
        aa += (a[i] ?? 0) ** 2;
      }
      if (power > 0) {
        const mix = (alpha * Math.sqrt(vv)) / Math.sqrt(aa);
        for (let i = 0; i < 3 * n; i++) v[i] = (1 - alpha) * (v[i] ?? 0) + mix * (a[i] ?? 0);
        if (++calm > 5) {
          dt = Math.min(dt * 1.1, maxDt);
          alpha *= 0.99;
        }
      } else {
        v.fill(0);
        dt *= 0.5;
        alpha = 0.1;
        calm = 0;
      }
      for (let i = 0; i < 3 * n; i += 3) {
        const vx = (v[i] ?? 0) + (a[i] ?? 0) * dt;
        const vy = (v[i + 1] ?? 0) + (a[i + 1] ?? 0) * dt;
        const vz = (v[i + 2] ?? 0) + (a[i + 2] ?? 0) * dt;
        const step = Math.sqrt(vx * vx + vy * vy + vz * vz) * dt;
        const cap = step > maxStep ? maxStep / step : 1;
        v[i] = vx * cap;
        v[i + 1] = vy * cap;
        v[i + 2] = vz * cap;
        x[i] += vx * cap * dt;
        x[i + 1] += vy * cap * dt;
        x[i + 2] += vz * cap * dt;
      }
    }
    return { iterations: maxIterations, settled: false, force };
  };

  const strain = (): number => {
    let worst = 0;
    for (const [p, q, length] of mesh.bars) {
      const len = Math.hypot(x[3 * q]! - x[3 * p]!, x[3 * q + 1]! - x[3 * p + 1]!, x[3 * q + 2]! - x[3 * p + 2]!);
      worst = Math.max(worst, Math.abs(len - length) / length);
    }
    return worst;
  };

  return {
    x,
    tracked,
    phase,
    restack: (stack: Stack) => touch.restack(stack),
    stats: touch.stats,
    strain,
    crossings: () => crossings(mesh, x, gap / 2),
  };
}
