#!/usr/bin/env bash
# Usage: tmp/tg/sweep.sh <name> [ENV=val ...] — relax and render into tmp/tg/<name>/,
# printing the run's summary. Runs in parallel when backgrounded.
set -euo pipefail
name=$1; shift
dir=tmp/tg/$name
mkdir -p "$dir"
env "$@" OUT="$dir/relaxed.json" npx tsx tmp/tg/dome.ts > "$dir/log" 2>&1
npx tsx tmp/tg/show.ts "$dir/relaxed.json" "$dir" >> "$dir/log" 2>&1
echo "== $name $*"
grep -E 'iters|start crossings' "$dir/log"
