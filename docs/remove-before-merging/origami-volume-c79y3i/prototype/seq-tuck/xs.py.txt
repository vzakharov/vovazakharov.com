"""xs.py <m.json> [y] — the strip's cross-section in the sim's x–y plane:
positions along paper x at paper y and 1-y (the two layers of the book fold),
plus each layer's distance from the base midline point's normal frame."""
import json, sys
d = json.load(open(sys.argv[1])); P = d['paper']; X = d['positions']
y = float(sys.argv[2]) if len(sys.argv) > 2 else 0.25


def near(c):
    return min(range(len(P)), key=lambda i: (P[i][0] - c[0]) ** 2 + (P[i][1] - c[1]) ** 2)


for yy in (y, 1 - y):
    print('paper y', yy)
    print('  ' + ' '.join('%.2f:(%.3f,%.3f)' % (x / 20, *X[near((x / 20, yy))][:2]) for x in range(21)))
