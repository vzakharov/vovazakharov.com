"""pt.py <m.json> 'px,py;qx,qy' ... — distance and positions of the vertices
nearest two paper points."""
import json, math, sys

d = json.load(open(sys.argv[1]))
P, X = d['paper'], d['positions']


def near(c):
    return min(range(len(P)), key=lambda i: (P[i][0] - c[0]) ** 2 + (P[i][1] - c[1]) ** 2)


def sub(p, q):
    return [p[k] - q[k] for k in range(3)]


def cross(u, w):
    return [u[1] * w[2] - u[2] * w[1], u[2] * w[0] - u[0] * w[2], u[0] * w[1] - u[1] * w[0]]


for pair in sys.argv[2:]:
    a, b = [list(map(float, s.split(','))) for s in pair.split(';')]
    i, j = near(a), near(b)
    # The second point's sheet normal, from its paper neighbours.
    e1 = sub(X[near([b[0] + 0.05, b[1]])], X[near([b[0] - 0.05, b[1]])])
    e2 = sub(X[near([b[0], b[1] + 0.05])], X[near([b[0], b[1] - 0.05])])
    n = cross(e1, e2)
    nn = math.hypot(*n)
    off = sub(X[i], X[j])
    normal = sum(off[k] * n[k] for k in range(3)) / nn
    print(pair, 'dist %.4f normal %.4f' % (math.dist(X[i], X[j]), normal))
