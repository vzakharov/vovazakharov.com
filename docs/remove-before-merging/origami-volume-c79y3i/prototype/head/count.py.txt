# count.py <steps.json> <n> x y : the layers (orders) at flat point x,y after n steps.
import json, subprocess, sys
out = subprocess.run(['npx', 'tsx', 'tmp/tg/head/flat.ts', sys.argv[1], sys.argv[2]], capture_output=True, text=True, cwd='/home/user/vovazakharov.com').stdout
pt = (float(sys.argv[3]), float(sys.argv[4]))
def inside(poly, p):
    s = 0
    for i in range(len(poly)):
        a, b = poly[i], poly[(i + 1) % len(poly)]
        c = (b[0] - a[0]) * (p[1] - a[1]) - (b[1] - a[1]) * (p[0] - a[0])
        if abs(c) < 1e-9: continue
        s |= 1 if c > 0 else 2
    return s != 3
ps = [json.loads(l) for l in out.split('\n') if l.startswith('{')]
print(len(ps), 'pieces; at', pt); [print(' o', p['o'], 'paper centroid', [round(sum(v[k] for v in p['paper'])/len(p['paper']),2) for k in (0,1)]) for p in ps if inside(p['flat'], pt)]
