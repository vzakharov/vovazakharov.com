#!/bin/bash
# go.sh <name> <steps-json-or-file> ; env as tmp/tg/lock/sweep.sh, with the
# lock settings defaulted. RENDER=1 renders and writes a half-size small.png.
cd /home/user/vovazakharov.com
out=tmp/tg/head/runs/head-$1
mkdir -p $out
START_REACH=${START_REACH:-1} FRICTION=${FRICTION:-0.5} EDGE_POINTS=${EDGE_POINTS:-3} EDGE_SET=${EDGE_SET:-outline} \
MAXIT=${MAXIT:-40000} timeout ${TO:-900} npx tsx tmp/tg/lock/spring.ts "$2" $out/m.json > $out/log.txt 2>&1
echo "[$1] exit $?"
grep -E "start crossings|^iters|nested-crease|real|pressed|strain 0|Error|error" $out/log.txt | grep -v "^  [0-9]*×" | head -${HEAD:-8}
if [ -n "$RENDER" ]; then ROT=${ROT:-0} timeout 300 npx tsx tmp/tg/show.ts $out/m.json $out >/dev/null 2>&1 && python3 -c "
from PIL import Image; im=Image.open('$out/views.png'); im.resize((im.width//2,im.height//2)).save('$out/small.png')" && echo rendered; fi
