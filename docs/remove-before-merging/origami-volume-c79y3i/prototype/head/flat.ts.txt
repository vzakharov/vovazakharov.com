// flat.ts <steps.json> [n] : prints the flat state after n steps (all by
// default) as JSON lines {o, flat, paper} for flat.py to draw.
import { readFileSync } from 'node:fs';
import { run, type Step } from '../lock/fold.ts';
import { flatPolygon } from '../../../src/origami/shared/fold/state.ts';
const steps = JSON.parse(readFileSync(process.argv[2]!, 'utf8')) as Step[];
const n = process.argv[3] ? Number(process.argv[3]) : steps.length;
const state = run(steps.slice(0, n));
for (const p of state.pieces.toSorted((a, b) => a.order - b.order)) {
  console.log(JSON.stringify({ o: p.order, flat: flatPolygon(p), paper: p.paper }));
}
