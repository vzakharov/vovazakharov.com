"""Where the paper's landmarks ended up: body frame, x along, y across, z up."""
import json, math, sys

d = json.load(open(sys.argv[1]))
P, pp = d["positions"], d["paper"]
zmin = min(p[2] for p in P)
r = -math.pi / 4


def body(p):
    u, v = p[0] - 0.5, p[1] - 0.5
    return (u * math.cos(r) - v * math.sin(r), u * math.sin(r) + v * math.cos(r), p[2] - zmin)


def at(x, y):
    i = min(range(len(pp)), key=lambda i: (pp[i][0] - x) ** 2 + (pp[i][1] - y) ** 2)
    return " ".join(f"{c:+.3f}" for c in body(P[i]))


xs = [body(p) for p in P]
print("extent x %.3f..%.3f y %.3f..%.3f z %.3f" % (min(b[0] for b in xs), max(b[0] for b in xs), min(b[1] for b in xs), max(b[1] for b in xs), max(b[2] for b in xs)))
for name, pt in {
    "centre": (0.5, 0.5),
    "corner00 (tucked tip)": (0, 0),
    "corner11 (tucked tip)": (1, 1),
    "corner10 (face tip)": (1, 0),
    "corner01 (face tip)": (0, 1),
    "foot .5,0": (0.5, 0),
    "foot 1,.5": (1, 0.5),
    "foot .5,1": (0.5, 1),
    "foot 0,.5": (0, 0.5),
}.items():
    print(f"{name:24} {at(*pt)}")
