import json, sys, copy
D = '/home/user/vovazakharov.com/tmp/tg/dilo3/'
BASE = json.load(open('/home/user/vovazakharov.com/tmp/tg/dilo/tf_long.steps.json'))
# indices: 0 spine, 1-2 inside (base sides), 3-4 front flaps raised, 5-6 back flaps (legs/tail),
# 7-8 front tips (frill), 9-10 back tips, 11 head, 12 jaw, 13 jaw pleat


def var(**h):
    s = copy.deepcopy(BASE)
    for k, v in h.items():
        s[int(k[1:])]["hold"] = v
    return s


V = {}
for a in (0.5, 0.6, 0.7, 0.9):
    V[f's{int(a*100)}'] = var(h0=a)


def mod(s, i, **kw):
    s[i] = {**s[i], **kw}
    return s


# slanted head line (crosses the spine off-perpendicular)
for a in (0.5, 0.6):
    t = int(a * 100)
    V[f'sl{t}_a'] = mod(var(h0=a), 11, line={"p": [0.32, 0.3], "q": [0.38, 0.7]})
    V[f'sl{t}_b'] = mod(var(h0=a), 11, line={"p": [0.30, 0.3], "q": [0.40, 0.7]})
    # head as a shaping fold, not flat
    for ang in (90, 120, 150):
        V[f'sh{t}_{ang}'] = mod(var(h0=a), 11, angle=ang)

# head by reverse fold of the apex (between back and belly), outside = wraps over
for a in (0.5, 0.6):
    t = int(a * 100)
    V[f'rvo{t}'] = mod(var(h0=a), 11, direction='outside', lower='L')
    V[f'rvi{t}'] = mod(var(h0=a), 11, direction='inside', lower='L')

# softer base sides with the spine override
for a, b in ((0.6, 0.8), (0.5, 0.8), (0.5, 0.7)):
    V[f's{int(a*100)}d{int(b*100)}'] = var(h0=a, h1=b, h2=b)
# softer head
for a in (0.5,):
    V[f's{int(a*100)}hd40'] = var(h0=a, h11=0.4)
for a in (0.5, 0.4):
    V[f's{int(a*100)}d80hd'] = var(h0=a, h1=0.8, h2=0.8, h11=0.4)
B = dict(h0=0.5, h1=0.8, h2=0.8, h11=0.4)
V['q_f45'] = var(**B, h3=0.45, h4=0.45)
V['q_f80'] = var(**B, h3=0.8, h4=0.8)
V['q_t60'] = var(**B, h7=0.6, h8=0.6)
V['q_t90'] = var(**B, h7=0.9, h8=0.9)
V['s50d80hd30'] =var(h0=0.5, h1=0.8, h2=0.8, h11=0.3)

for n in (sys.argv[1:] or V):
    json.dump(V[n], open(D + n + '.steps.json', 'w'))
    print(n)
