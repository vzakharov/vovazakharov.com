// Symmetric degree-6 waterbomb vertex at the origin: a midline along x (rays
// of length m to ±x), four diagonals of length L at 45°, 135°, 225°, 315°. The
// sectors 45°–135° and 225°–315° stay flat (the big faces); the 45° sectors
// fold in pairs about the midline rays. Big face 1 is tilted by φ about x.
type V = [number, number, number];
const sub = (a: V, b: V): V => [a[0] - b[0], a[1] - b[1], a[2] - b[2]];
const dot = (a: V, b: V) => a[0] * b[0] + a[1] * b[1] + a[2] * b[2];
const unit = (a: V): V => {
  const n = Math.sqrt(dot(a, a));
  return [a[0] / n, a[1] / n, a[2] / n];
};
/** Fold angle across edge o→p between triangles (o,p,a) and (o,p,b): 0 flat. */
function fold(o: V, p: V, a: V, b: V): number {
  const e = unit(sub(p, o));
  const perp = (q: V): V => {
    const d = sub(q, o);
    const s = dot(d, e);
    return unit([d[0] - s * e[0], d[1] - s * e[1], d[2] - s * e[2]]);
  };
  return Math.PI - Math.acos(Math.max(-1, Math.min(1, dot(perp(a), perp(b)))));
}
const L = 0.5;
const k = L / Math.SQRT2;
const m = k;
const deg = (r: number) => ((r * 180) / Math.PI).toFixed(1);
for (const phiDeg of [100, 110, 120, 130, 140, 150]) {
  const phi = (phiDeg * Math.PI) / 180;
  const d45: V = [k, k * Math.cos(phi), -k * Math.sin(phi)];
  const d315: V = [k, k * Math.cos(phi), k * Math.sin(phi)];
  // M = (p, 0, q): p² + q² = m², |M − d45|² = k² (k = m here).
  let best: V = [0, 0, 0];
  let err = Infinity;
  for (let s = 0; s < 2 * Math.PI; s += 1e-5) {
    const mm: V = [m * Math.cos(s), 0, m * Math.sin(s)];
    const d = sub(mm, d45);
    const e = Math.abs(dot(d, d) - k * k);
    // the tucked solution: M not on the big faces' own continuation
    if (e < err && mm[0] < k * 0.999) {
      err = e;
      best = mm;
    }
  }
  const o: V = [0, 0, 0];
  const diag = fold(o, d45, [-k, k * Math.cos(phi), -k * Math.sin(phi)], best);
  const mid = fold(o, best, d45, d315);
  // apex height over the feet, and the feet's spread
  console.log(
    `φ=${phiDeg} M=${best.map((v) => v.toFixed(3))} err=${err.toExponential(1)} diag=${deg(diag)} mid=${deg(mid)} feet z=${(-d45[2]).toFixed(3)} spanY=${(2 * d45[1]).toFixed(3)}`,
  );
}
