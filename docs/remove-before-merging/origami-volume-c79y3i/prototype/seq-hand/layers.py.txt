# layers.py <m.json> : at flat points near the top and bottom free edges, each
# layer's position (nearest vertex) for the maw's base: right outer (o0), right
# reversed (o1 top / o2 bottom), left reversed (o4 top / o3 bottom), front (o5).
import json, sys, math
d = json.load(open(sys.argv[1]))
def at(p):
    i = min(range(len(d['paper'])), key=lambda k: math.dist(d['paper'][k], p))
    return d['positions'][i], math.dist(d['paper'][i], p)
for u, v in [(0.02, 0.58), (0.05, 0.62), (0.1, 0.7), (0.2, 0.6)]:
    for name, w, layers in [('top', v, [('o0', (1 - u, v)), ('o1R', (v, 1 - u)), ('o4L', (1 - v, 1 - u)), ('o5F', (u, v))]),
                            ('bot', 1 - v, [('o0', (1 - u, 1 - v)), ('o2R', (v, u)), ('o3L', (1 - v, u)), ('o5F', (u, 1 - v))])]:
        row = []
        for n, p in layers:
            x, e = at(p)
            row.append(f'{n}({p[0]:.2f},{p[1]:.2f}) {x[0]:.4f},{x[1]:.4f},{x[2]*1e3:.2f}e-3')
        print(f'{name} flat({u},{w:.2f}):', ' | '.join(row))
