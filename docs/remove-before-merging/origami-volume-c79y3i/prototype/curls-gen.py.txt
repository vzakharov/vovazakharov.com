import json, math, sys
def fan(cx, cy, angles, holds, dirs=None, r=0.3):
    steps=[]
    for i,(a,h) in enumerate(zip(angles,holds)):
        t=math.radians(a); m=math.radians(a-8)
        d = dirs[i] if dirs else ("valley" if i%2==0 else "mountain")
        steps.append({"line":{"p":[cx,cy],"q":[round(cx+0.4*math.cos(t),12),round(cy+0.4*math.sin(t),12)]},
          "moving":[round(cx+r*math.cos(m),12),round(cy+r*math.sin(m),12)],"select":"all","direction":d,"hold":h})
    return steps
if __name__=="__main__":
    print(json.dumps(eval(sys.argv[1])))
