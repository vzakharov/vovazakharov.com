// pass.ts <steps.json> : per step, the still layers stacked between a moved
// piece and the still piece it is hinged to across the crease, covering the
// crease — layers the flap's turn would have to pass through.
import { readFileSync } from 'node:fs';

import { add, apply, contains, overlaps, perpendicular, scale } from '../../../src/origami/shared/fold/geometry.ts';
import { run, type Step } from './fold.ts';

const steps = JSON.parse(readFileSync(process.argv[2]!, 'utf8')) as Step[];
for (let k = 1; k <= steps.length; k++) {
  const s = run(steps, k);
  const { moved, creases, line } = s.log[k - 1]!;
  const n = scale(perpendicular(line.d), 1e-4 / Math.hypot(...line.d));
  const through = new Map<string, number>();
  for (const [a, b] of creases) {
    for (const t of [0.25, 0.5, 0.75]) {
      const mid = [a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t] as const;
      const mover = s.pieces.find((p) => moved.has(p.id) && contains(p.paper, mid));
      const still = s.pieces.find((p) => !moved.has(p.id) && contains(p.paper, mid));
      if (!mover || !still) continue;
      const flat = apply(still.placement, mid);
      const [lo, hi] = [mover.order, still.order].toSorted((x, y) => x - y);
      for (const p of s.pieces) {
        if (moved.has(p.id) || p === still || p.order <= lo! || p.order >= hi!) continue;
        const poly = p.paper.map((q) => apply(p.placement, q));
        if (contains(poly, add(flat, n)) && contains(poly, add(flat, scale(n, -1)))) {
          const key = `o${p.order}(id${p.id}) between o${mover.order}/o${still.order}`;
          through.set(key, (through.get(key) ?? 0) + 1);
        }
      }
    }
  }
  console.log(`step ${k} ${steps[k - 1]!.direction}: ${through.size ? [...through.keys()].join(', ') : 'clear'}`);
  // The swept side: before the step, a still layer over (valley) or under
  // (mountain) a moved piece where that piece starts — one the turn lifts it through.
  const dir = steps[k - 1]!.direction;
  if (dir !== 'valley' && dir !== 'mountain') continue;
  const pre = run(steps, k - 1);
  const postMoved = s.pieces.filter((p) => moved.has(p.id));
  const centre = (poly: readonly (readonly [number, number])[]) => [poly.reduce((a, q) => a + q[0], 0) / poly.length, poly.reduce((a, q) => a + q[1], 0) / poly.length] as const;
  const swept = new Set<string>();
  for (const b of pre.pieces) {
    const parts = postMoved.filter((p) => contains(b.paper, centre(p.paper)));
    for (const part of parts) {
      const start = part.paper.map((q) => apply(b.placement, q));
      for (const st of pre.pieces) {
        if (st === b || postMoved.some((p) => contains(st.paper, centre(p.paper)))) continue;
        const above = st.order > b.order;
        if ((dir === 'valley') !== above) continue;
        if (overlaps(start, st.paper.map((q) => apply(st.placement, q)))) swept.add(`id${st.id}(o${st.order}) ${above ? 'over' : 'under'} id${b.id}(o${b.order})`);
      }
    }
  }
  if (swept.size) console.log(`  swept through: ${[...swept].slice(0, 8).join(', ')}${swept.size > 8 ? ` (+${swept.size - 8})` : ''}`);
}
