/**
 * Lets a folded mesh spring back and settle, as real paper does once let go:
 * stiff bars hold every edge at its paper length, creases pull toward the angle
 * they keep, plain paper toward flat, more softly — which is what lets it bend
 * rather than crease — and layers keep a sheet's thickness apart.
 *
 * Explicit damped dynamics as in Origami Simulator (Ghassaei et al. 2018),
 * damped by FIRE (Bitzek et al. 2006), on the CPU and deterministic.
 */
import { contact, crossings } from './contact.ts';
import { dihedral } from '../../../src/origami/shared/fold/hinge.ts';
import type { Mesh } from '../../../src/origami/shared/fold/mesh.ts';
import type { Paper } from '../../../src/origami/shared/fold/paper.ts';
import type { Vec3 } from '../../../src/origami/shared/fold/space.ts';

/** Stiffnesses relative to one another; only their ratios shape the result. */
const AXIAL = 20;
const CREASE = Number(process.env.CREASE ?? 0.7);
const FACET = Number(process.env.FACET ?? 0.7);
/** Contact, per unit of overlap into the gap, scaled by the mesh edge. */
const CONTACT = 4 * AXIAL;
/** Contacts a vertex can carry at once: its own, and those of the vertices
 * over every triangle around it. */
const CONTACT_LOAD = 12;

/** Settled once no vertex feels more than this: it moves a bar by ~1e-8 of
 * the sheet, and contact switching on and off keeps the last digits from
 * calming further. */
const TOLERANCE = 1e-5;
const MAX_ITERATIONS = Number(process.env.MAXIT ?? 200_000);
/** Steps between rebuilding the contact pairs. */
const REFRESH = 10;
/** Steps over which the hinges' goals move to where they rest. */
const RAMP = 2000;

export type Relaxation = {
  /** Every mesh vertex in space, settled. */
  readonly positions: readonly Vec3[];
  readonly iterations: number;
  /** The most any edge stretched or shrank, as a share of its paper length. */
  readonly strain: number;
  /** Mesh edges left passing through paper they are not joined to, by more
   * than half its thickness. */
  readonly crossings: number;
  readonly report?: readonly string[];
};

/** The model's edges and hinges packed for the force loop. */
function pack(mesh: Mesh, x: Float64Array) {
  const bars = Int32Array.from(mesh.bars.flatMap(([a, b]) => [a, b]));
  const rest = Float64Array.from(mesh.bars.map(([, , length]) => length));
  const hinges = mesh.hinges;
  const corners = Int32Array.from(hinges.flatMap((h) => [h.a, h.b, h.c, h.d]));
  const stiffness = Float64Array.from(
    hinges.map((h) => (h.crease ? CREASE : FACET) * h.rest),
  );
  const target = Float64Array.from(hinges.map((h) => h.angle));
  // The angle is tracked across steps, so a crease folded flat keeps its side
  // past ±π.
  const tracked = Float64Array.from(
    hinges.map((h) => unwrap(dihedral(x, h), h.start)),
  );
  return { bars, rest, corners, stiffness, target, tracked };
}

/** `angle` moved by whole turns to lie nearest `near`. */
const unwrap = (angle: number, near: number): number =>
  angle + 2 * Math.PI * Math.round((near - angle) / (2 * Math.PI));

export function relax(mesh: Mesh, paper: Paper): Relaxation {
  const n = mesh.paper.length;
  const gap = paper.thickness / paper.side;
  const edge = Math.max(...mesh.bars.map(([, , length]) => length));
  const x = Float64Array.from(mesh.start.flat());
  const v = new Float64Array(3 * n);
  const f = new Float64Array(3 * n);
  const a = new Float64Array(3 * n);
  const gradient = new Float64Array(12);
  const { bars, rest, corners, stiffness, target, tracked } = pack(mesh, x);
  // Each hinge's goal moves from where it starts to where it rests over the
  // first steps, so the paper opens near balance rather than flung open.
  const from = Float64Array.from(tracked);
  let ramp = 0;
  const wings = { a: 0, b: 0, c: 0, d: 0 };
  const touch = contact(mesh, gap, CONTACT / edge);

  // The stiffest any vertex is held bounds the step: Gershgorin over the bars,
  // the hinges and a contact on each side.
  // The stick spring of friction holds a vertex as stiffly again, sideways.
  const rubbing = process.env.FRICTION || process.env.BOND ? 1 + Number(process.env.SHEAR ?? 1) : 1;
  const held = new Float64Array(n).fill((CONTACT_LOAD * CONTACT * rubbing) / edge);
  for (const [a, b, length] of mesh.bars) {
    for (const i of [a, b]) held[i] = (held[i] ?? 0) + (2 * AXIAL) / length;
  }
  for (const [h, hinge] of mesh.hinges.entries()) {
    dihedral(x, hinge, gradient);
    for (const [k, i] of [hinge.a, hinge.b, hinge.c, hinge.d].entries()) {
      const g2 =
        (gradient[3 * k] ?? 0) ** 2 +
        (gradient[3 * k + 1] ?? 0) ** 2 +
        (gradient[3 * k + 2] ?? 0) ** 2;
      held[i] = (held[i] ?? 0) + 4 * (stiffness[h] ?? 0) * g2;
    }
  }
  // Each vertex weighs what holds it, so every one rings at the same top
  // frequency and the step no longer waits on the stiffest corner.
  const mass = held;
  const maxDt = 1.8;
  /** No vertex moves farther in one step than half the gap, so no layer
   * steps through another unseen. */
  const maxStep = gap / 2;

  const forces = (): number => {
    f.fill(0);
    for (let e = 0; e < rest.length; e++) {
      const a = 3 * (bars[2 * e] ?? 0);
      const b = 3 * (bars[2 * e + 1] ?? 0);
      const dx = (x[b] ?? 0) - (x[a] ?? 0);
      const dy = (x[b + 1] ?? 0) - (x[a + 1] ?? 0);
      const dz = (x[b + 2] ?? 0) - (x[a + 2] ?? 0);
      const len = Math.sqrt(dx * dx + dy * dy + dz * dz);
      const length = rest[e] ?? len;
      const pull = (AXIAL * (len - length)) / length / len;
      f[a] = (f[a] ?? 0) + pull * dx;
      f[a + 1] = (f[a + 1] ?? 0) + pull * dy;
      f[a + 2] = (f[a + 2] ?? 0) + pull * dz;
      f[b] = (f[b] ?? 0) - pull * dx;
      f[b + 1] = (f[b + 1] ?? 0) - pull * dy;
      f[b + 2] = (f[b + 2] ?? 0) - pull * dz;
    }
    for (let h = 0; h < target.length; h++) {
      wings.a = corners[4 * h] ?? 0;
      wings.b = corners[4 * h + 1] ?? 0;
      wings.c = corners[4 * h + 2] ?? 0;
      wings.d = corners[4 * h + 3] ?? 0;
      const angle = unwrap(dihedral(x, wings, gradient), tracked[h] ?? 0);
      tracked[h] = angle;
      const goal = (from[h] ?? 0) + ((target[h] ?? 0) - (from[h] ?? 0)) * ramp;
      const torque = -(stiffness[h] ?? 0) * (angle - goal);
      for (let k = 0; k < 4; k++) {
        const i = 3 * (corners[4 * h + k] ?? 0);
        f[i] = (f[i] ?? 0) + torque * (gradient[3 * k] ?? 0);
        f[i + 1] = (f[i + 1] ?? 0) + torque * (gradient[3 * k + 1] ?? 0);
        f[i + 2] = (f[i + 2] ?? 0) + torque * (gradient[3 * k + 2] ?? 0);
      }
    }
    touch.push(x, f);
    let worst = 0;
    for (let i = 0; i < 3 * n; i += 3) {
      const fx = f[i] ?? 0;
      const fy = f[i + 1] ?? 0;
      const fz = f[i + 2] ?? 0;
      worst = Math.max(worst, fx * fx + fy * fy + fz * fz);
    }
    return Math.sqrt(worst);
  };

  let dt = maxDt / 10;
  let alpha = 0.1;
  let calm = 0;
  for (let iteration = 0; iteration < MAX_ITERATIONS; iteration++) {
    ramp = Math.min(1, iteration / RAMP);
    if (iteration % REFRESH === 0) touch.refresh(x, 2 * REFRESH * maxStep);
    if (forces() < TOLERANCE && ramp === 1) {
      return { ...settled(mesh, x, iteration, gap), report: [`settled; rubbing ${JSON.stringify(touch.stats)}`] };
    }
    for (let i = 0; i < 3 * n; i++) {
      a[i] = (f[i] ?? 0) / (mass[Math.floor(i / 3)] ?? 1);
    }
    let power = 0;
    let vv = 0;
    let aa = 0;
    for (let i = 0; i < 3 * n; i++) {
      const ai = a[i] ?? 0;
      const vi = v[i] ?? 0;
      power += ai * vi;
      vv += vi * vi;
      aa += ai * ai;
    }
    if (power > 0) {
      const mix = (alpha * Math.sqrt(vv)) / Math.sqrt(aa);
      for (let i = 0; i < 3 * n; i++) {
        v[i] = (1 - alpha) * (v[i] ?? 0) + mix * (a[i] ?? 0);
      }
      if (++calm > 5) {
        dt = Math.min(dt * 1.1, maxDt);
        alpha *= 0.99;
      }
    } else {
      v.fill(0);
      dt *= 0.5;
      alpha = 0.1;
      calm = 0;
    }
    for (let i = 0; i < 3 * n; i += 3) {
      const vx = (v[i] ?? 0) + (a[i] ?? 0) * dt;
      const vy = (v[i + 1] ?? 0) + (a[i + 1] ?? 0) * dt;
      const vz = (v[i + 2] ?? 0) + (a[i + 2] ?? 0) * dt;
      const step = Math.sqrt(vx * vx + vy * vy + vz * vz) * dt;
      const cap = step > maxStep ? maxStep / step : 1;
      v[i] = vx * cap;
      v[i + 1] = vy * cap;
      v[i + 2] = vz * cap;
      x[i] = (x[i] ?? 0) + vx * cap * dt;
      x[i + 1] = (x[i + 1] ?? 0) + vy * cap * dt;
      x[i + 2] = (x[i + 2] ?? 0) + vz * cap * dt;
    }
  }
  const left = forces();
  return { ...settled(mesh, x, MAX_ITERATIONS, gap), report: [`UNSETTLED force ${left.toExponential(2)}; rubbing ${JSON.stringify(touch.stats)}`] };
}

function settled(
  mesh: Mesh,
  x: Float64Array,
  iterations: number,
  gap: number,
): Relaxation {
  let strain = 0;
  for (const [a, b, length] of mesh.bars) {
    const len = Math.hypot(
      (x[3 * b] ?? 0) - (x[3 * a] ?? 0),
      (x[3 * b + 1] ?? 0) - (x[3 * a + 1] ?? 0),
      (x[3 * b + 2] ?? 0) - (x[3 * a + 2] ?? 0),
    );
    strain = Math.max(strain, Math.abs(len - length) / length);
  }
  return {
    positions: Array.from(
      { length: mesh.paper.length },
      (_, i): Vec3 => [x[3 * i] ?? 0, x[3 * i + 1] ?? 0, x[3 * i + 2] ?? 0],
    ),
    iterations,
    strain,
    crossings: crossings(mesh, x, gap / 2),
  };
}
