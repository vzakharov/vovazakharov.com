// Renders tmp/tg/relaxed.json through the origami renderer, one piece per
// mesh triangle.
import { readFileSync } from 'node:fs';
import path from 'node:path';

import { render } from '../../src/origami/scripts/lib/render.ts';

type Vec3 = readonly [number, number, number];
const data = JSON.parse(readFileSync(process.argv[2] ?? 'tmp/tg/relaxed.json', 'utf8')) as {
  triangles: [number, number, number][];
  paper: [number, number][];
  positions: Vec3[];
  faces: { facing: number }[];
};
const zs = data.positions.map((p) => p[2]);
const flip = process.env.FLIP ? -1 : 1;
const lo = Math.min(...zs.map((z) => z * flip));
const pieces = data.triangles.map((t, i) => ({
  id: i,
  paper: t.map((v) => data.paper[v]!),
  world: t.map((v) => {
    const [x, y, z] = data.positions[v]!;
    return [x, flip * y, flip * z - lo] as Vec3;
  }),
  front: (data.faces[i]?.facing ?? 1) > 0,
  layer: 0,
}));
const model = { pieces, creases: [] };
const views = [
  { name: 'three-quarter', az: -35, el: 30, zoom: 1.05 },
  { name: 'side', az: 0, el: 8, zoom: 1.05 },
  { name: 'head-on', az: 90, el: 12, zoom: 1.05 },
  { name: 'top', az: 0, el: 90, zoom: 1.05 },
];
const out = path.resolve(process.argv[3] ?? 'tmp/tg/out');
await render(
  {
    steps: [],
    colors: { front: 0x8a_9a_5b, back: 0xe8_dc_c0 },
    display: (p) => p,
  },
  out,
  [{ frames: views.map((view) => ({ model, view })), file: path.join(out, 'views.png') }],
);
console.log('wrote', out);
