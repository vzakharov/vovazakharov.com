// Re-poses a relaxed model in the pressed creature's frame: forward = paper
// (0,0.5)→(0.5,0.5) direction, up = the front normal of the back (left face
// around x∈[0.08,0.3]), so views compare with the pressed dinosaur.
// Usage: npx tsx tmp/tg/dilo2/orient.ts <in.json> <out.json>
import { readFileSync, writeFileSync } from 'node:fs';

type V = [number, number, number];
const d = JSON.parse(readFileSync(process.argv[2]!, 'utf8')) as {
  triangles: [number, number, number][];
  paper: [number, number][];
  positions: V[];
  faces: { facing: number }[];
};
const sub = (a: V, b: V): V => [a[0] - b[0], a[1] - b[1], a[2] - b[2]];
const cross = (a: V, b: V): V => [a[1] * b[2] - a[2] * b[1], a[2] * b[0] - a[0] * b[2], a[0] * b[1] - a[1] * b[0]];
const dot = (a: V, b: V) => a[0] * b[0] + a[1] * b[1] + a[2] * b[2];
const unit = (a: V): V => { const n = Math.hypot(...a); return [a[0] / n, a[1] / n, a[2] / n]; };
const near = (x: number, y: number) => {
  let best = 0, bd = 9;
  d.paper.forEach((p, i) => { const e = Math.hypot(p[0] - x, p[1] - y); if (e < bd) { bd = e; best = i; } });
  return d.positions[best]!;
};
let up: V = [0, 0, 0];
for (const t of d.triangles) {
  const c = t.map((v) => d.paper[v]!);
  const cx = (c[0]![0] + c[1]![0] + c[2]![0]) / 3, cy = (c[0]![1] + c[1]![1] + c[2]![1]) / 3;
  if (cx < 0.08 || cx > 0.3 || Math.abs(cy - 0.5) > 0.12) continue;
  const turn = (c[1]![0] - c[0]![0]) * (c[2]![1] - c[0]![1]) - (c[1]![1] - c[0]![1]) * (c[2]![0] - c[0]![0]);
  const [a, b, e] = t.map((v) => d.positions[v]!);
  const n = cross(sub(b!, a!), sub(e!, a!));
  const s = Math.sign(turn);
  up = [up[0] + s * n[0], up[1] + s * n[1], up[2] + s * n[2]];
}
up = unit(up);
const tail = near(0, 0.5), apex = near(0.5, 0.5);
let f = sub(apex, tail);
f = unit(sub(f, [up[0] * dot(f, up), up[1] * dot(f, up), up[2] * dot(f, up)]));
const side = cross(up, f);
const o = apex;
d.positions = d.positions.map((p) => {
  const q = sub(p, o);
  return [0.5 + dot(q, f), 0.5 + dot(q, side), dot(q, up)];
});
writeFileSync(process.argv[3]!, JSON.stringify(d));
