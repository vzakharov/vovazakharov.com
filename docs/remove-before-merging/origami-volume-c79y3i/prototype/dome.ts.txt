export function fromFlat(m: Mesh): Mesh {
  return {
    ...m,
    start: m.paper.map(([x, y]) => [x, y, 0] as const),
    sheets: m.sheets.map(() => []),
  };
}

/** The mesh of the final state started from an earlier one, `pose`: every
 * vertex where the pose put it, each crease the pose had not yet folded flat
 * at 0, the contact stack read off the pose. */
export function fromPose(m: Mesh, pose: FoldState): Mesh {
  const lift = paper.thickness / paper.side;
  const layerOf = layers(pose);
  const holding = (pt: Vec) =>
    pose.pieces.filter((p) => contains(p.paper, pt));
  const start = m.paper.map((pt): [number, number, number] => {
    const around = holding(pt);
    const sum = around.reduce(
      (s, p) => {
        const at = transform(p.shaping, [
          ...apply(p.placement, pt),
          (layerOf.get(p.id) ?? 0) * lift,
        ]);
        return [s[0] + at[0], s[1] + at[1], s[2] + at[2]];
      },
      [0, 0, 0],
    );
    return [sum[0] / around.length, sum[1] / around.length, sum[2] / around.length];
  });
  const centroid = (ids: readonly number[]): Vec => {
    const pts = ids.map((i) => m.paper[i] ?? [0, 0]);
    return [
      pts.reduce((s, p) => s + p[0], 0) / pts.length,
      pts.reduce((s, p) => s + p[1], 0) / pts.length,
    ];
  };
  const pieceAt = (pt: Vec) => {
    const found = holding(pt)[0];
    if (!found) throw new Error(`no pose piece at ${pt.join(',')}`);
    return found;
  };
  const x = Float64Array.from(start.flat());
  const hinges = m.hinges.map((h) => {
    const one = pieceAt(centroid([h.a, h.b, h.c]));
    const two = pieceAt(centroid([h.b, h.a, h.d]));
    const folded = one !== two && Math.abs(one.order - two.order) > 0;
    const geometric = dihedral(x, h);
    return { ...h, start: folded ? Math.sign(h.start) * Math.PI : geometric };
  });
  const stack = (p: Piece) => ({ order: p.order, plane: 0 });
  return {
    ...m,
    start,
    hinges,
    faces: m.triangles.map((t) => {
      const p = pieceAt(centroid(t));
      return { ...stack(p), facing: Math.sign(determinant(p.placement)) };
    }),
    sheets: m.paper.map((pt) => holding(pt).map(stack)),
  };
}

type V3 = [number, number, number];
const cross = (a: V3, b: V3): V3 => [
  a[1] * b[2] - a[2] * b[1],
  a[2] * b[0] - a[0] * b[2],
  a[0] * b[1] - a[1] * b[0],
];
const dot3 = (a: V3, b: V3) => a[0] * b[0] + a[1] * b[1] + a[2] * b[2];
const norm = (a: V3): V3 => {
  const n = Math.sqrt(dot3(a, a));
  return [a[0] / n, a[1] / n, a[2] / n];
};

/** The base's one vertex, folded rigidly: big faces tilted by ψ about the
 * midline (0 flat, 90° pressed), the 45° sector pairs tucked between them.
 * Built in the frame e1 = midline, e2 across it, e3 the pose's up. Returns,
 * for a pose-flat point, its 3D place and the side it faces. */
export function waterbombVertex(psi: number, tent: number) {
  const c = 0.5;
  const r2 = Math.SQRT1_2;
  const s = -2 * Math.atan(Math.sin(psi));
  const sp = tent * Math.sin(psi);
  const cp = Math.cos(psi);
  const Mplus: V3 = [Math.cos(s), 0, tent * Math.sin(s)];
  const Mminus: V3 = [-Math.cos(s), 0, tent * Math.sin(s)];
  // Big face A (v > 0) and B (v < 0): rotations about e1.
  const faceA = (a: number, b: number): V3 => [a, b * cp, -b * sp];
  const faceB = (a: number, b: number): V3 => [a, b * cp, b * sp];
  /** The rigid map taking flat unit `u0` to `M` and flat unit `d0` to `D`. */
  const sector = (u0: [number, number], M: V3, d0: [number, number], D: V3) => {
    // flat frame: u0, w0 ⟂ u0 toward d0, normal +z
    const sgn = Math.sign(u0[0] * d0[1] - u0[1] * d0[0]);
    const w0: [number, number] = [-sgn * u0[1], sgn * u0[0]];
    const w = norm([D[0] - dot3(D, M) * M[0], D[1] - dot3(D, M) * M[1], D[2] - dot3(D, M) * M[2]]);
    const n = sgn > 0 ? cross(M, w) : cross(w, M);
    return (a: number, b: number, z: number): V3 => {
      const p = a * u0[0] + b * u0[1];
      const q = a * w0[0] + b * w0[1];
      return [p * M[0] + q * w[0] + z * n[0], p * M[1] + q * w[1] + z * n[1], p * M[2] + q * w[2] + z * n[2]];
    };
  };
  const dAp: V3 = [r2, r2 * cp, -r2 * sp]; // (u+v)/√2 on face A
  const dAm: V3 = [-r2, r2 * cp, -r2 * sp]; // (−u+v)/√2 on face A
  const dBp: V3 = [r2, -r2 * cp, -r2 * sp]; // (u−v)/√2 on face B
  const dBm: V3 = [-r2, -r2 * cp, -r2 * sp]; // (−u−v)/√2 on face B
  const S2 = sector([1, 0], Mplus, [r2, r2], dAp);
  const S1 = sector([1, 0], Mplus, [r2, -r2], dBp);
  const S3 = sector([-1, 0], Mminus, [-r2, r2], dAm);
  const S4 = sector([-1, 0], Mminus, [-r2, -r2], dBm);
  const nA: V3 = [0, sp, cp];
  const nB: V3 = [0, -sp, cp];
  return (x: number, y: number, z: number): V3 => {
    // (a, b) along the midline and across it
    const a = ((x - c) + (y - c)) * r2;
    const b = (-(x - c) + (y - c)) * r2;
    const ang = Math.atan2(b, a); // 0 along +u
    const q = Math.PI / 4;
    let p: V3;
    if (ang >= q && ang <= 3 * q) {
      const f = faceA(a, b);
      p = [f[0] + z * nA[0], f[1] + z * nA[1], f[2] + z * nA[2]];
    } else if (ang <= -q && ang >= -3 * q) {
      const f = faceB(a, b);
      p = [f[0] + z * nB[0], f[1] + z * nB[1], f[2] + z * nB[2]];
    } else if (ang >= 0 && ang < q) p = S2(a, b, z);
    else if (ang < 0 && ang > -q) p = S1(a, b, z);
    else if (ang > 3 * q) p = S3(a, b, z);
    else p = S4(a, b, z);
    // back to world: e1 = u, e2 = v, e3 = z
    const ux = r2;
    return [c + (p[0] - p[1]) * ux, c + (p[0] + p[1]) * ux, p[2]];
  };
}

/** `fromPose`, then every point carried rigidly to the folded vertex, and each
 * base crease made to rest where it starts. */
export function fromVertex(m: Mesh, pose: FoldState, psi: number, tent: number): Mesh {
  const posed = fromPose(m, pose);
  const at = waterbombVertex(psi, tent);
  const start = posed.start.map(([x, y, z]) => at(x, y, z));
  const x = Float64Array.from(start.flat());
  const hinges = posed.hinges.map((h, i) => {
    const pressed = m.hinges[i]!;
    const base = Math.abs(h.start) < 3; // not a flap folded flat in the pose
    if (!h.crease) return { ...h, start: dihedral(x, h) };
    if (!base) return h;
    const now = dihedral(x, h);
    const hold = Number(process.env.KEEP ?? 1);
    return { ...h, start: now, angle: now * hold + 0 * pressed.angle };
  });
  return { ...posed, start, hinges };
}

if (process.argv[1]?.endsWith('dome.ts')) {
  const state = run(steps);
  console.log('pieces', state.pieces.length);
  if (process.env.LIST) {
    for (const p of state.pieces) {
      console.log(
        p.id,
        p.order,
        [...p.tags].join(''),
        flatPolygon(p)
          .map((v) => v.map((n) => n.toFixed(3)).join(','))
          .join(' '),
      );
    }
  }
  const pressed = mesh(state, paper);
  const pose = process.env.POSE;
  const m = process.env.PSI
    ? fromVertex(pressed, run(steps, PRE), (Number(process.env.PSI) * Math.PI) / 180, Number(process.env.TENT ?? 1))
    : process.env.FLAT
    ? fromFlat(pressed)
    : pose
      ? fromPose(pressed, run(steps, Number(pose)))
      : pressed;
  const gap = paper.thickness / paper.side;
  const flipSet = new Set((process.env.FLIPSET ?? '').split(',').filter(Boolean));
  if (flipSet.size) {
    const on = (h: (typeof m.hinges)[number]) => {
      const [p, q] = [m.paper[h.a]!, m.paper[h.b]!];
      const mx = (p[0] + q[0]) / 2;
      const my = (p[1] + q[1]) / 2;
      const ex = (a: number, b: number) => Math.abs(a - b) < 1e-6;
      if (ex(p[0], 0.5) && ex(q[0], 0.5)) return my > 0.5 ? 'xhi' : 'xlo';
      if (ex(p[1], 0.5) && ex(q[1], 0.5)) return mx > 0.5 ? 'yhi' : 'ylo';
      if (ex(p[0], p[1]) && ex(q[0], q[1])) return mx > 0.5 ? 'mhi' : 'mlo';
      return '';
    };
    (m as { hinges: typeof m.hinges }).hinges = m.hinges.map((h) =>
      h.crease && flipSet.has(on(h)) ? { ...h, angle: -h.angle } : h,
    );
  }
  console.log(
    'verts',
    m.paper.length,
    'hinges',
    m.hinges.length,
    'start crossings',
    crossings(m, Float64Array.from(m.start.flat()), gap / 2),
  );
  const t0 = Date.now();
  const r = relax(m, paper);
  console.log(
    'iters',
    r.iterations,
    'strain',
    r.strain,
    'crossings',
    r.crossings,
    'ms',
    Date.now() - t0,
  );
  if (process.env.DIAG) {
    const xr = Float64Array.from(r.positions.flat());
    const groups = new Map<string, { n: number; got: number; worst: number; at: string }>();
    for (const h of m.hinges) {
      if (!h.crease) continue;
      const got = dihedral(xr, h);
      const key = `${h.crease ? 'C' : 'F'} target ${((h.angle * 180) / Math.PI).toFixed(1)}`;
      const g = groups.get(key) ?? { n: 0, got: 0, worst: 0, at: '' };
      g.n++;
      g.got += (got * 180) / Math.PI;
      const dev = Math.abs(got - h.angle);
      if (dev > g.worst) {
        g.worst = dev;
        g.at = [h.a, h.b].map((i) => m.paper[i]!.map((v) => v.toFixed(2)).join(',')).join(' ');
      }
      groups.set(key, g);
    }
    const seg = new Map<string, string[]>();
    for (const h of m.hinges) {
      if (!h.crease) continue;
      const [p, q] = [m.paper[h.a]!, m.paper[h.b]!];
      const mid = [(p[0] + q[0]) / 2, (p[1] + q[1]) / 2];
      const line = Math.abs(p[0] - p[1]) < 1e-6 && Math.abs(q[0] - q[1]) < 1e-6 ? 'y=x' : Math.abs(p[0] - 0.5) < 1e-6 && Math.abs(q[0] - 0.5) < 1e-6 ? 'x=.5' : Math.abs(p[1] - 0.5) < 1e-6 && Math.abs(q[1] - 0.5) < 1e-6 ? 'y=.5' : '';
      if (!line) continue;
      const key = `${line} ${mid[0]! < 0.5 ? 'lo' : 'hi'}${mid[1]! < 0.5 ? 'lo' : 'hi'} ${mid[0]! > 0.25 && mid[0]! < 0.75 && mid[1]! > 0.25 && mid[1]! < 0.75 ? 'base' : 'flap'}`;
      const s = `${((h.angle * 180) / Math.PI).toFixed(0)}→${((dihedral(xr, h) * 180) / Math.PI).toFixed(0)}`;
      seg.set(key, [...(seg.get(key) ?? []), s]);
    }
    for (const [k, v] of [...seg].sort()) console.log(k, v.join(' '));
    for (const [k, g] of groups)

  }

    process.env.OUT ?? 'tmp/tg/relaxed.json',
    JSON.stringify({ triangles: m.triangles, paper: m.paper, positions: r.positions, faces: m.faces }),
  );
}
