export function fromFlat(m: Mesh): Mesh {
  return {
    ...m,
    start: m.paper.map(([x, y]) => [x, y, 0] as const),
    sheets: m.sheets.map(() => []),
  };
}

/** The mesh of the final state started from an earlier one, `pose`: every
 * vertex where the pose put it, each crease the pose had not yet folded flat
 * at 0, the contact stack read off the pose. */
export function fromPose(m: Mesh, pose: FoldState): Mesh {
  const lift = paper.thickness / paper.side;
  const layerOf = layers(pose);
  const holding = (pt: Vec) =>
    pose.pieces.filter((p) => contains(p.paper, pt));
  const start = m.paper.map((pt): [number, number, number] => {
    const around = holding(pt);
    const sum = around.reduce(
      (s, p) => {
        const at = transform(p.shaping, [
          ...apply(p.placement, pt),
          (layerOf.get(p.id) ?? 0) * lift,
        ]);
        return [s[0] + at[0], s[1] + at[1], s[2] + at[2]];
      },
      [0, 0, 0],
    );
    return [sum[0] / around.length, sum[1] / around.length, sum[2] / around.length];
  });
  const centroid = (ids: readonly number[]): Vec => {
    const pts = ids.map((i) => m.paper[i] ?? [0, 0]);
    return [
      pts.reduce((s, p) => s + p[0], 0) / pts.length,
      pts.reduce((s, p) => s + p[1], 0) / pts.length,
    ];
  };
  const pieceAt = (pt: Vec) => {
    const found = holding(pt)[0];
    if (!found) throw new Error(`no pose piece at ${pt.join(',')}`);
    return found;
  };
  const x = Float64Array.from(start.flat());
  const hinges = m.hinges.map((h) => {
    const one = pieceAt(centroid([h.a, h.b, h.c]));
    const two = pieceAt(centroid([h.b, h.a, h.d]));
    const folded = one !== two && Math.abs(one.order - two.order) > 0;
    const geometric = dihedral(x, h);
    return { ...h, start: folded ? Math.sign(h.start) * Math.PI : geometric };
  });
  const stack = (p: Piece) => ({ order: p.order, plane: 0 });
  return {
    ...m,
    start,
    hinges,
    faces: m.triangles.map((t) => {
      const p = pieceAt(centroid(t));
      return { ...stack(p), facing: Math.sign(determinant(p.placement)) };
    }),
    sheets: m.paper.map((pt) => holding(pt).map(stack)),
  };
}

if (process.argv[1]?.endsWith('dome.ts')) {
  const state = run(steps);
  console.log('pieces', state.pieces.length);
  if (process.env.LIST) {
    for (const p of state.pieces) {
      console.log(
        p.id,
        p.order,
        [...p.tags].join(''),
        flatPolygon(p)
          .map((v) => v.map((n) => n.toFixed(3)).join(','))
          .join(' '),
      );
    }
  }
  const pressed = mesh(state, paper);
  const pose = process.env.POSE;
  const m = process.env.FLAT
    ? fromFlat(pressed)
    : pose
      ? fromPose(pressed, run(steps, Number(pose)))
      : pressed;
  const gap = paper.thickness / paper.side;
  console.log(
    'verts',
    m.paper.length,
    'hinges',
    m.hinges.length,
    'start crossings',
    crossings(m, Float64Array.from(m.start.flat()), gap / 2),
  );
  const t0 = Date.now();
  const r = relax(m, paper);
  console.log(
    'iters',
    r.iterations,
    'strain',
    r.strain,
    'crossings',
    r.crossings,
    'ms',
    Date.now() - t0,
  );
  writeFileSync(
    'tmp/tg/relaxed.json',
    JSON.stringify({ triangles: m.triangles, paper: m.paper, positions: r.positions, faces: m.faces }),
  );
}
