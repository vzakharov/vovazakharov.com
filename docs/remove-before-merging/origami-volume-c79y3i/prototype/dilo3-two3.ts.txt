// two.ts (dilo2) plus per-segment hold overrides, applied after the mesh is built:
//   SPINE="near,far,r0,r1"  spine hinges (paper x=0.5) hold `near` where |y-.5|<r0,
//                           `far` where |y-.5|>r1, linear between.
//   OVR='[{"box":[x0,y0,x1,y1],"hold":h}]'  crease hinges with both ends in the box.
// Also prints where the worst strain sits, in paper.
// Usage: npx tsx tmp/tg/dilo3/two3.ts <steps.json> <out.json>   env BASEN (3), BPOSE
import { readFileSync, writeFileSync } from 'node:fs';

import { run, type Step } from '../../../src/origami/shared/fold/fold.ts';
import { mesh, type Mesh } from '../../../src/origami/shared/fold/mesh.ts';
import { crossings } from '../../../src/origami/shared/fold/contact.ts';
import { dihedral } from '../../../src/origami/shared/fold/hinge.ts';
import { PAPERS } from '../../../src/origami/shared/fold/paper.ts';
import { relax } from '../relax.ts';
import { fromPose } from '../dome.ts';

const paper = PAPERS['kami-15'];
const steps = JSON.parse(readFileSync(process.argv[2]!, 'utf8')) as Step[];
const baseN = Number(process.env.BASEN ?? 3);
const gap = paper.thickness / paper.side;

const baseSteps = steps.slice(0, baseN);
const bm = fromPose(mesh(run(baseSteps), paper), run(baseSteps, 0));
const br = relax(bm, paper);
console.log('base strain', br.strain.toFixed(4), 'crossings', br.crossings);

function locate(pt: readonly number[]): [number, number, number] {
  for (const [a, b, c] of bm.triangles) {
    const A = bm.paper[a]!, B = bm.paper[b]!, C = bm.paper[c]!;
    const d = (B[1] - C[1]) * (A[0] - C[0]) + (C[0] - B[0]) * (A[1] - C[1]);
    const l1 = ((B[1] - C[1]) * (pt[0]! - C[0]) + (C[0] - B[0]) * (pt[1]! - C[1])) / d;
    const l2 = ((C[1] - A[1]) * (pt[0]! - C[0]) + (A[0] - C[0]) * (pt[1]! - C[1])) / d;
    const l3 = 1 - l1 - l2;
    if (l1 >= -1e-9 && l2 >= -1e-9 && l3 >= -1e-9) {
      const P = br.positions[a]!, Q = br.positions[b]!, R = br.positions[c]!;
      return [0, 1, 2].map((k) => l1 * P[k]! + l2 * Q[k]! + l3 * R[k]!) as [number, number, number];
    }
  }
  throw new Error(`no base triangle at ${pt.join(',')}`);
}

const full = mesh(run(steps), paper);
const posed = fromPose(full, run(steps, Number(process.env.BPOSE ?? baseN)));
const start = full.paper.map(locate);
const x = Float64Array.from(start.flat());

const spine = process.env.SPINE?.split(',').map(Number);
const radial = process.env.RADIAL?.split(',').map(Number);
const ovr = JSON.parse(process.env.OVR ?? '[]') as { box: [number, number, number, number]; hold: number }[];
const P = full.paper;
let touched = 0;
const hinges = posed.hinges.map((h) => {
  let angle = h.angle;
  if (h.crease) {
    const a = P[h.a]!, b = P[h.b]!;
    const fullAngle = h.start;
    if (spine && Math.abs(a[0] - 0.5) < 1e-6 && Math.abs(b[0] - 0.5) < 1e-6) {
      const [near, far, r0, r1] = spine as [number, number, number, number];
      const dy = Math.abs((a[1] + b[1]) / 2 - 0.5);
      const t = dy <= r0 ? 0 : dy >= r1 ? 1 : (dy - r0) / (r1 - r0);
      angle = (near + (far - near) * t) * fullAngle;
      touched++;
    }
    // RADIAL="near,r0,r1": every base crease (spine x=.5, diagonals through the
    // centre) pressed to `near` within r0 of the centre, its own hold past r1.
    if (radial) {
      const onBase = (p: readonly number[]) =>
        Math.abs(p[0]! - 0.5) < 1e-6 || Math.abs(Math.abs(p[0]! - 0.5) - Math.abs(p[1]! - 0.5)) < 1e-6;
      if (onBase(a) && onBase(b)) {
        const [near, r0, r1] = radial as [number, number, number];
        const r = Math.hypot((a[0] + b[0]) / 2 - 0.5, (a[1] + b[1]) / 2 - 0.5);
        const t = r <= r0 ? 0 : r >= r1 ? 1 : (r - r0) / (r1 - r0);
        const own = angle / fullAngle;
        angle = (near + (own - near) * t) * fullAngle;
        touched++;
      }
    }
    for (const o of ovr) {
      const [x0, y0, x1, y1] = o.box;
      const inside = (p: readonly number[]) => p[0]! >= x0 && p[0]! <= x1 && p[1]! >= y0 && p[1]! <= y1;
      if (inside(a) && inside(b)) { angle = o.hold * fullAngle; touched++; }
    }
  }
  return { ...h, angle, start: dihedral(x, h) };
});
console.log('overridden hinges', touched);
const m: Mesh = { ...posed, start, hinges };
console.log('start crossings', crossings(m, x, gap / 2));
const r = relax(m, paper);
// Where the worst-stretched bars are, in paper.
const worst = m.bars
  .map(([a, b, len]) => {
    const p = r.positions[a]!, q = r.positions[b]!;
    const l = Math.hypot(p[0] - q[0], p[1] - q[1], p[2] - q[2]);
    return { s: Math.abs(l - len) / len, at: [(P[a]![0] + P[b]![0]) / 2, (P[a]![1] + P[b]![1]) / 2] };
  })
  .toSorted((u, v) => v.s - u.s)
  .slice(0, 4)
  .map((w) => `${(w.s * 100).toFixed(1)}%@(${w.at.map((c) => c.toFixed(2)).join(',')})`);
console.log('iters', r.iterations, 'strain', r.strain.toFixed(4), 'crossings', r.crossings, 'worst', worst.join(' '));
writeFileSync(
  process.argv[3]!,
  JSON.stringify({ triangles: m.triangles, paper: m.paper, positions: r.positions, faces: m.faces }),
);
