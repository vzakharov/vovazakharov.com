// Contact sheet of the generator's survivors: one three-quarter view each.
// Usage: npx tsx tmp/tg/sheet.ts <dir> [maxCrossings] [maxStrain] [minVolume] [limit]
import { readFileSync, readdirSync } from 'node:fs';
import path from 'node:path';

import { render } from '../../src/origami/scripts/lib/render.ts';

type Vec3 = readonly [number, number, number];
type Row = { seed: number; fail?: string; crossings: number; strain: number; volume: number };
const dir = process.argv[2] ?? 'tmp/tg/gen';
const maxX = Number(process.argv[3] ?? 0);
const maxS = Number(process.argv[4] ?? 0.03);
const minV = Number(process.argv[5] ?? 0.2);
const limit = Number(process.argv[6] ?? 24);
const rows: Row[] = readdirSync(dir)
  .filter((f) => f.startsWith('results') && f.endsWith('.jsonl'))
  .flatMap((f) => readFileSync(path.join(dir, f), 'utf8').trim().split('\n'))
  .filter(Boolean)
  .map((l) => JSON.parse(l) as Row);
const ok = rows
  .filter((r) => !r.fail && r.crossings <= maxX && r.strain <= maxS && r.volume >= minV)
  .toSorted((a, b) => b.volume - a.volume)
  .slice(Number(process.env.OFF ?? 0), Number(process.env.OFF ?? 0) + limit);
console.log(`${rows.length} tried, ${rows.filter((r) => !r.fail).length} relaxed, ${ok.length} shown`);

function model(seed: number) {
  const data = JSON.parse(readFileSync(path.join(dir, `${seed}.json`), 'utf8')) as {
    triangles: [number, number, number][];
    paper: [number, number][];
    positions: Vec3[];
    faces: { facing: number }[];
  };
  const lo = Math.min(...data.positions.map((p) => p[2]));
  const ccw = (t: [number, number, number]): [number, number, number] => {
    const [a, b, c] = t.map((v) => data.paper[v]!);
    const turn = (b![0] - a![0]) * (c![1] - a![1]) - (b![1] - a![1]) * (c![0] - a![0]);
    return turn > 0 ? t : [t[0], t[2], t[1]];
  };
  return {
    pieces: data.triangles.map(ccw).map((t, i) => ({
      id: i,
      paper: t.map((v) => data.paper[v]!),
      world: t.map((v) => {
        const [x, y, z] = data.positions[v]!;
        return [x, y, z - lo] as Vec3;
      }),
      front: (data.faces[i]?.facing ?? 1) > 0,
      layer: 0,
    })),
    creases: [],
  };
}

const out = path.resolve(dir, 'sheet');
const az = Number(process.env.AZ ?? -35);
await render(
  { steps: [], colors: { front: 0x8a_9a_5b, back: 0xe8_dc_c0 }, display: (p) => p },
  out,
  [
    {
      frames: ok.map((r) => ({
        model: model(r.seed),
        view: {
          name: `#${r.seed} v${r.volume.toFixed(2)} s${(100 * r.strain).toFixed(1)}% x${r.crossings}`,
          az,
          el: 30,
          zoom: 1.05,
        },
      })),
      file: path.join(out, `sheet-az${az}-${process.env.OFF ?? 0}.png`),
    },
  ],
);
console.log('wrote', out);
