/**
 * Flies legs through `steer` the way `InsectView` does, with a perch whose
 * end moves (a cap wobbling from a tap, or beckoning) and legs flown in from
 * off screen, and reads them the way `scripts/lib/flier-watch.ts` does.
 *
 *   node --import tsx tmp/handle-bite6/heading/sim.ts [seeds]
 */

import type { Point } from '../../../src/pages/mushrooms/model/geometry.ts';
import type { InsectKind } from '../../../src/pages/mushrooms/model/insect-genes.ts';
import { wrap } from '../../../src/pages/mushrooms/model/insect-motion.ts';
import {
  type Fluttering,
  PATH_SHAPES,
} from '../../../src/pages/mushrooms/model/insect-paths.ts';
import {
  startLeg,
  steer,
  type Steering,
} from '../../../src/pages/mushrooms/model/insect-steering.ts';
import { mulberry32 } from '../../../src/pages/mushrooms/model/random.ts';

const FRAME = 1000 / 60;
const FLYING: Record<InsectKind, [number, number]> = {
  butterfly: [2400, 3900],
  fly: [600, 1100],
  bee: [1100, 1800],
};

/** A cap's top at `t` ms: its foot, its height, stretched and rocked by a tap's bounce or a beckon. */
type Cap = { foot: Point; height: number; across: number; tappedAt: number; litAt: number };

function bounce(elapsed: number): number {
  if (elapsed < 0 || elapsed >= 1.92) return 0;
  return -0.2 * Math.exp(-2.4 * elapsed) * Math.cos(Math.PI * 2 * 1.8 * elapsed);
}
function beckon(elapsed: number): number {
  if (elapsed < 0) return 0;
  return 0.06 * Math.min(1, elapsed / 0.3) * Math.sin(((Math.PI * 2) / 1.3) * elapsed);
}
function capTop(cap: Cap, moving: number): Point {
  const now = process.env.STILL ? 0 : moving;
  const t = now / 1000;
  const b = bounce(t - cap.tappedAt / 1000);
  const stretch = 0.018 * Math.sin(((Math.PI * 2) / 3.4) * t) + b + beckon(t - cap.litAt / 1000);
  const [x, y] = [cap.across * (1 - stretch / 2), -cap.height * (1 + stretch)];
  const turn = b * 0.35;
  return {
    x: cap.foot.x + x * Math.cos(turn) - y * Math.sin(turn),
    y: cap.foot.y + x * Math.sin(turn) + y * Math.cos(turn),
  };
}

type Worst = { off: number; spin: number; at: string; spun: string };

/** One leg as the view flies it, watched as the flier watch reads it. */
function flyLeg(
  motion: Fluttering,
  screen: { width: number; height: number },
  start: Point,
  sat: number | undefined,
  cap: Cap,
  flight: number,
  size: number,
  worst: Worst,
  label: string,
): void {
  const span = size * { butterfly: 0.9, fly: 1.1, bee: 1 }[motion.kind];
  const window = 2 * Math.round(30 / PATH_SHAPES[motion.kind].flutterRate);
  const leg = { departs: 0, arrives: flight, to: { kind: 'cap', id: 'c' } as const };
  let steering: Steering = startLeg({ facing: 0, turn: sat ?? 0, at: -Infinity, setOff: undefined, perch: undefined });
  const carried = { launch: 0, speed: 0, drink: 0 };
  let aim: Point | undefined;
  const trail: { x: number; y: number; turn: number; now: number }[] = [];
  let [round, least, most] = [0, 0, 0];
  for (let now = 0; now < flight + 400; now += FRAME) {
    const end = capTop(cap, now);
    aim ??= end;
    const last = steering.turn;
    const flown = steer(
      steering,
      { leg, carried, start, end, aim, sat, perched: true, size, motion },
      now,
    );
    steering = flown.steering;
    if (Number.isFinite(last)) round += wrap(steering.turn - last);
    [least, most] = [Math.min(least, round), Math.max(most, round)];
    if (most - least > worst.spin) [worst.spin, worst.spun] = [most - least, label];
    const { x, y } = flown.point;
    if (process.env.ALL === `${label} ${motion.kind} ${String(screen.width)}`) {
      console.log(`${(now - flight).toFixed(0)} end ${end.x.toFixed(1)},${end.y.toFixed(1)} at ${x.toFixed(1)},${y.toFixed(1)} facing ${steering.facing.toFixed(2)} turn ${steering.turn.toFixed(2)} round ${round.toFixed(2)} setOff ${JSON.stringify(steering.setOff)}`);
    }
    trail.push({ x, y, turn: steering.turn, now });
    if (trail.length <= window) continue;
    trail.shift();
    const [first, middle] = [trail[0], trail[window / 2]];
    if (!first || !middle) continue;
    const travel = Math.hypot(x - first.x, y - first.y);
    const onScreen = x >= 0 && x <= screen.width && y >= 0 && y <= screen.height;
    if (
      middle.now < leg.departs + 150 ||
      now >= leg.arrives ||
      !onScreen ||
      travel < (span * (now - first.now)) / 1000
    ) {
      continue;
    }
    const off = Math.abs(wrap(middle.turn - Math.PI / 2 - Math.atan2(y - first.y, x - first.x)));
    if (process.env.TRACE === `${label} ${motion.kind} ${String(screen.width)}`) {
      console.log(
        `${(flight - now).toFixed(0)} end ${end.x.toFixed(1)},${end.y.toFixed(1)} at ${x.toFixed(1)},${y.toFixed(1)} way ${Math.atan2(y - first.y, x - first.x).toFixed(2)} face@mid ${(middle.turn - Math.PI / 2).toFixed(2)} facing ${steering.facing.toFixed(2)} off ${off.toFixed(2)} travel ${travel.toFixed(1)}`,
      );
    }
    if (off > worst.off) {
      worst.off = off;
      worst.at = `${label} ${(flight - now).toFixed(0)} ms before landing, travel ${travel.toFixed(1)}`;
    }
  }
}

const seeds = Number(process.argv[2] ?? 60);
const screens = [
  { name: 'tabL', width: 1180, height: 820, size: 104 },
  { name: 'tabP', width: 820, height: 1180, size: 104 },
  { name: 'phoneP', width: 390, height: 844, size: 60 },
  { name: 'phoneL', width: 844, height: 390, size: 60 },
];
for (const [scenario, kind] of [
  ['hop', 'fly'],
  ['hop', 'bee'],
  ['same', 'butterfly'],
  ['same', 'fly'],
  ['same', 'bee'],
  ['in', 'butterfly'],
  ['in', 'fly'],
] as const) {
  for (const screen of screens) {
    const worst: Worst = { off: 0, spin: 0, at: '', spun: '' };
    const size = screen.size * { butterfly: 1, fly: 0.55, bee: 0.65 }[kind];
    for (let seed = 1; seed <= seeds; seed += 1) {
      const random = mulberry32(seed * 7919 + 13);
      const motion = { phase: random() * Math.PI * 2, kind, flutter: size * 0.28 };
      const height = size * (1.5 + random() * 2);
      const foot = {
        x: screen.width * (0.15 + 0.7 * random()),
        y: screen.height * (0.6 + 0.3 * random()) + height,
      };
      const cap: Cap = {
        foot,
        height,
        across: (random() - 0.5) * size,
        tappedAt: scenario === 'in' ? -1e9 : random() * 60 - 30,
        litAt: random() < 0.5 ? random() * -1000 : -1e9,
      };
      const [lo, hi] = FLYING[kind];
      const flight = lo + (hi - lo) * random();
      const label = `${scenario} seed ${String(seed)}`;
      if (scenario === 'hop') {
        const seat = capTop(cap, -1e6);
        const angle = random() * Math.PI * 2;
        const reach = size * (0.4 + random());
        const start = { x: seat.x + Math.cos(angle) * reach, y: seat.y + Math.sin(angle) * reach };
        flyLeg(motion, screen, start, (random() - 0.5) * 0.9, cap, flight, size, worst, label);
      } else if (scenario === 'same') {
        const start = capTop({ ...cap, tappedAt: -1e9 }, 0);
        flyLeg(motion, screen, start, (random() - 0.5) * 0.9, cap, flight, size, worst, label);
      } else {
        const right = random() < 0.5;
        const start = {
          x: right ? screen.width + size * 1.5 : -size * 1.5,
          y: screen.height * (0.18 + 0.32 * random()),
        };
        flyLeg(motion, screen, start, undefined, cap, flight, size, worst, label);
      }
    }
    console.log(
      `${scenario} ${kind.padEnd(9)} ${screen.name.padEnd(6)} worst ${worst.off.toFixed(2)} rad (${worst.at}), spin ${(worst.spin / Math.PI / 2).toFixed(2)} turns (${worst.spun})`,
    );
  }
}
